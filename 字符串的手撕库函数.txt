/*题目五：字符串的"手撕库函数"
任务：在不使用 <string.h> 任何函数的前提下，自己实现三个函数：

int myStrlen(const char *s) —— 计算字符串长度

void myStrcpy(char *dst, const char *src) —— 字符串拷贝

int myStrcmp(const char *s1, const char *s2) —— 字符串比较

然后用一个测试程序验证它们和标准库函数行为一致。*/
#include <stdio.h>
#include <string.h>
int mystrlen(const char *s){
    int idx=0;
    while (s[idx]!='\0'){idx++;}
    return idx;
}  
char* mystrcpy(char *dst, const char *src){
    char *ret = dst;
    while (*src!='\0'){*dst=*src;dst++;src++;}
        *dst='\0';
        return ret;
}
int mystrcmp(const char *s1, const char *s2){
    while(*s1==*s2&&*s1!='\0'){
        s1++;s2++;
    }
    return *s1-*s2;
}
int main(){
    char s1[32]={"hello,my name is Aearone."};
    char s2[32]={"Hello,my name is Aearone."};
    char s3[32]={"abc"};
    char s4[32]={"abc"};
    printf("字符长度为%d\n",mystrlen(s1));
    printf("字符长度为%d\n",strlen(s1));
    printf("字符相差为%d\n",mystrcmp(s1,s2));
    printf("字符相差为%d\n",strcmp(s1,s2));
    printf("%s\n",mystrcpy(s3,s1));
    printf("%s\n",strcpy(s4,s1));
    return 0;
}
