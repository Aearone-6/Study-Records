/* 编写一个函数 int isAgentNumber(int n)，判断一个数是否为“特工数”。
定义：一个 n 位数，如果每位数字的 n 次方之和等于它本身，就是特工数。*/

#include <stdio.h>
#include <math.h>
int isAgentNumber(int n) {
    int temp = n;
    int digit=0;
    // 计算数字的位数
    while (temp > 0) {
        digit++;
        temp /= 10;
    }
    temp = n;
    int sum = 0;
    // 计算每位数字的 n 次方之和
    while(temp>0){
        int digitValue = temp % 10;
        int power=1;
        for(int i=0;i<digit;i++){
            power *= digitValue;
        }
        sum += power;
        temp /= 10;
    }
    // 判断是否为特工数
    return sum == n;    
}
int main() {
    int number;
    printf("请输入一个整数: ");
    scanf("%d", &number);
    
    if (isAgentNumber(number)) {
        printf("%d 是特工数。\n", number);
    } else {
        printf("%d 不是特工数。\n", number);
    }
    
    return 0;
}
